    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    function playTone(freq, type = 'sine', duration = 0.18) {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
      gain.gain.setValueAtTime(0.18, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + duration);
    }

    function flipCard(card) {
      if (card.classList.contains('flipped') || card.classList.contains('matched') || flippedCards.length === 2) return;
      playTone(440, 'sine', 0.1);
      card.classList.add('flipped'); card.textContent = card.dataset.emoji;
      flippedCards.push(card);
      if (flippedCards.length === 2) {
        const [c1, c2] = flippedCards;
        if (c1.dataset.emoji === c2.dataset.emoji) {
          setTimeout(() => {
            c1.classList.add('matched'); c2.classList.add('matched');
            flippedCards = []; matchedCount++;
            playTone(660, 'triangle', 0.15);
            if (matchedCount === animals.length) {

            c1.classList.remove('flipped'); c2.classList.remove('flipped');

    function setBrushSize(size, btn) {

    function initCanvas() {
      const canvas = document.getElementById('drawCanvas');
      const rect = canvas.getBoundingClientRect();
      if (!canvasReady) {
        canvas.width = rect.width;
        canvas.height = 270;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = '#FFFFFF';
        ctx.fillRect(0, 0, canvas.width, canvas.height);
      }
      const ctx = canvas.getContext('2d');
      ctx.lineCap = 'round'; ctx.lineJoin = 'round';

      const paletteColors = ['#EF4444', '#F97316', '#FACC15', '#22C55E', '#3B82F6', '#A855F7', '#EC4899', '#1E293B'];
      const row = document.getElementById('paletteRow');
      row.innerHTML = '';
      paletteColors.forEach(col => {
        const dot = document.createElement('div');
        dot.className = 'swatch' + (col === brushColor ? ' active' : '');
        dot.style.background = col;
        dot.onclick = () => {
          brushColor = col;
          document.querySelectorAll('.swatch').forEach(s => s.classList.remove('active'));
          dot.classList.add('active');
          playTone(520, 'sine', 0.08);
        };
        row.appendChild(dot);
      });

      if (canvasReady) return;
      canvasReady = true;

      const getPos = (e) => {
        const r = canvas.getBoundingClientRect();
        const clientX = e.touches ? e.touches[0].clientX : e.clientX;
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: clientX - r.left, y: clientY - r.top };
      };
      const startDraw = (e) => { isDrawing = true; const p = getPos(e); ctx.beginPath(); ctx.moveTo(p.x, p.y); };
      const draw = (e) => {
        if (!isDrawing) return;
        e.preventDefault();
        const p = getPos(e);
        ctx.lineWidth = brushSize; ctx.strokeStyle = brushColor;
        ctx.lineTo(p.x, p.y); ctx.stroke();
      };
      const stopDraw = () => { isDrawing = false; };

      canvas.addEventListener('mousedown', startDraw);
      canvas.addEventListener('mousemove', draw);
      window.addEventListener('mouseup', stopDraw);
      canvas.addEventListener('touchstart', startDraw, { passive: false });
      canvas.addEventListener('touchmove', draw, { passive: false });
      canvas.addEventListener('touchend', stopDraw);
    }

    function clearCanvas() {
      const c = document.getElementById('drawCanvas');
      const ctx = c.getContext('2d');
      ctx.fillStyle = '#FFFFFF';
      ctx.fillRect(0, 0, c.width, c.height);

    function drawTemplate(type) {
      clearCanvas();
      if (type === 'none') return;
      const c = document.getElementById('drawCanvas');
      const ctx = c.getContext('2d');
      const cx = c.width / 2, cy = c.height / 2;
      ctx.strokeStyle = '#94A3B8';
      ctx.lineWidth = 4;

      if (type === 'house') {
        ctx.strokeRect(cx - 70, cy - 10, 140, 100);
        ctx.beginPath(); ctx.moveTo(cx - 85, cy - 10); ctx.lineTo(cx, cy - 85); ctx.lineTo(cx + 85, cy - 10); ctx.closePath(); ctx.stroke();
        ctx.strokeRect(cx - 20, cy + 35, 40, 55);
      } else if (type === 'sun') {
        ctx.beginPath(); ctx.arc(cx, cy, 50, 0, Math.PI * 2); ctx.stroke();
        for (let i = 0; i < 8; i++) {
          const a = (i * 45) * Math.PI / 180;
          ctx.beginPath(); ctx.moveTo(cx + 62 * Math.cos(a), cy + 62 * Math.sin(a));
          ctx.lineTo(cx + 90 * Math.cos(a), cy + 90 * Math.sin(a)); ctx.stroke();
        }
      } else if (type === 'fish') {
        ctx.beginPath(); ctx.ellipse(cx - 10, cy, 75, 45, 0, 0, Math.PI * 2); ctx.stroke();
        ctx.beginPath(); ctx.moveTo(cx + 65, cy); ctx.lineTo(cx + 110, cy - 35); ctx.lineTo(cx + 110, cy + 35); ctx.closePath(); ctx.stroke();
        ctx.beginPath(); ctx.arc(cx - 45, cy - 10, 6, 0, Math.PI * 2); ctx.stroke();
      }
      playTone(540, 'sine', 0.1);
    }

      { name: 'Mary Had a Little Lamb', seq: [2, 1, 0, 1, 2, 2, 2, 1, 1, 1, 2, 4, 4] }
    ];
    let guideActive = false, curSongIdx = 0, guideIndex = 0;

    function initPiano() {
      guideActive = false; guideIndex = 0;
      document.getElementById('songStatus').textContent = '🎵 Mod Bebas';
      renderPianoKeys();
    }

    function startSong(idx) {
      curSongIdx = idx; guideActive = true; guideIndex = 0;
      document.getElementById('songStatus').textContent = `⭐ Main: ${songs[idx].name}`;
      renderPianoKeys();
    }

    function renderPianoKeys() {
      const grid = document.getElementById('pianoGrid');
      grid.innerHTML = '';
      pianoNotes.forEach(note => {
        const btn = document.createElement('button');
        btn.className = 'piano-key';
        if (guideActive && songs[curSongIdx].seq[guideIndex] === note.id) btn.classList.add('guide-glow');
        btn.style.background = note.color;
        btn.style.boxShadow = `0 6px 0 ${note.dark}`;
        btn.innerHTML = `<span>${note.emoji}</span>${note.label}`;
        btn.onclick = () => {
          playTone(note.freq, 'triangle', 0.35);
          if (guideActive && songs[curSongIdx].seq[guideIndex] === note.id) {
            guideIndex++;
            if (guideIndex >= songs[curSongIdx].seq.length) {
              guideActive = false;
              document.getElementById('songStatus').textContent = '🎉 Lagu Selesai!';
              playSuccessSound();
              addStar(2, 'Pandai Main Piano! +2 ⭐', 'general');
            }
            renderPianoKeys();
          }
        };
        grid.appendChild(btn);
      });
    }

    updateUI();
